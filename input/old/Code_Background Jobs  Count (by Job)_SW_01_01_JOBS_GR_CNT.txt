FUNCTION /SKN/F_SW_01_01_JOBS_GRP_CNT.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     VALUE(IS_ALERT) TYPE  CHAR1
*"  TABLES
*"      T_SELECT STRUCTURE  RSSELECT OPTIONAL
*"      T_DATA STRUCTURE  /SKN/S_SW_01_01_JOBS_GRP_CNT OPTIONAL
*"----------------------------------------------------------------------
DATA : LV_ALERT TYPE  CHAR1.
DATA : LS_DET TYPE /SKN/S_SW_01_01_JOBS,
       LT_DET LIKE TABLE OF LS_DET.
DATA : LV_CNT TYPE I.
DATA : LS_DATA LIKE LINE OF T_DATA,
       LT_DATA LIKE TABLE OF LS_DATA.
DATA_MULTY: JOBS_CNT /SKN/E_SW_CNT.
SELECT_MULTY: JOBS_CNT.
***data_single: AGGR_PERIOD CHAR1.
***select_single: AGGR_PERIOD.
   REFRESH T_DATA.
   CALL FUNCTION '/SKN/F_SW_01_01_JOBS_N'
    IMPORTING
       IS_ALERT       = LV_ALERT
     TABLES
       T_SELECT       = T_SELECT
       T_DATA         = LT_DET.
    SORT LT_DET BY JOBNAME VARIANT JOBCOUNT STRTDATE STRTTIME.
    DELETE ADJACENT DUPLICATES FROM LT_DET COMPARING JOBNAME VARIANT JOBCOUNT STRTDATE STRTTIME.
    LOOP AT LT_DET INTO LS_DET.
      MOVE-CORRESPONDING LS_DET TO LS_DATA.
      LS_DATA-JOBS_CNT = 1.
      COLLECT LS_DATA INTO LT_DATA.
    ENDLOOP.
    DELETE LT_DATA WHERE JOBS_CNT NOT IN R_JOBS_CNT.
    T_DATA[] = LT_DATA[].
    IF T_DATA[] IS NOT INITIAL.
      IS_ALERT = 'X'.
    ENDIF.
ENDFUNCTION.
