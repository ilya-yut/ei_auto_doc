FUNCTION /SKN/F_SW_01_01_JOBS_GRP_DET.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     VALUE(IS_ALERT) TYPE  CHAR1
*"  TABLES
*"      T_SELECT STRUCTURE  RSSELECT OPTIONAL
*"      T_DATA STRUCTURE  /SKN/S_SW_01_01_JOBS_GRP_DET OPTIONAL
*"----------------------------------------------------------------------
DATA: LV_ALERT TYPE  CHAR1.
DATA: LS_DET TYPE /SKN/S_SW_01_01_JOBS,
      LT_DET LIKE TABLE OF LS_DET.
DATA: LV_CNT TYPE I.
DATA: LS_DATA LIKE LINE OF T_DATA,
      LT_DATA LIKE TABLE OF LS_DATA.
DATA: LS_GRP_CNT TYPE /SKN/S_SW_01_01_JOBS_GRP_CNT,
      LT_GRP_CNT LIKE TABLE OF LS_GRP_CNT.
DATA_MULTY: JOBS_CNT /SKN/E_SW_CNT.
SELECT_MULTY: JOBS_CNT.
DATA_MULTY: TOT_JOBNAME BTCJOB.
   REFRESH T_DATA.
   CALL FUNCTION '/SKN/F_SW_01_01_JOBS_N'
    IMPORTING
       IS_ALERT       = LV_ALERT
     TABLES
       T_SELECT       = T_SELECT
       T_DATA         = LT_DET.
    SORT LT_DET BY JOBNAME VARIANT JOBCOUNT STRTDATE STRTTIME.
    DELETE ADJACENT DUPLICATES FROM LT_DET COMPARING JOBNAME VARIANT JOBCOUNT STRTDATE STRTTIME.
    LOOP AT LT_DET INTO LS_DET.
      MOVE-CORRESPONDING LS_DET TO LS_GRP_CNT.
      LS_GRP_CNT-JOBS_CNT = 1.
      COLLECT LS_GRP_CNT INTO LT_GRP_CNT.
    ENDLOOP.
    DELETE LT_GRP_CNT WHERE JOBS_CNT NOT IN R_JOBS_CNT.
    CHECK LT_GRP_CNT[] IS NOT INITIAL.
    SORT LT_GRP_CNT BY JOBNAME VARIANT.
***    refresh r_TOT_JOBNAME.
***    loop at lt_GRP_CNT into ls_GRP_CNT.
***      rs_TOT_JOBNAME-sign = 'I'.
***      rs_TOT_JOBNAME-option = 'EQ'.
***      rs_TOT_JOBNAME-low = ls_GRP_CNT-JOBNAME.
***      append rs_TOT_JOBNAME to r_TOT_JOBNAME.
***    endloop.
***    delete lt_det where JOBNAME not in r_TOT_JOBNAME.
    LOOP AT LT_DET INTO LS_DET.
      READ TABLE LT_GRP_CNT INTO LS_GRP_CNT WITH KEY JOBNAME = LS_DET-JOBNAME
                                                     VARIANT = LS_DET-VARIANT
                                            BINARY SEARCH.
      IF SY-SUBRC IS INITIAL.
        MOVE-CORRESPONDING LS_DET TO LS_DATA.
        APPEND LS_DATA TO LT_DATA.
      ENDIF.
    ENDLOOP.
    T_DATA[] = LT_DATA[].
    IF T_DATA[] IS NOT INITIAL.
      IS_ALERT = 'X'.
    ENDIF.
ENDFUNCTION.
