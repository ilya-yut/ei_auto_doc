FUNCTION /SKN/F_SW_01_01_VERS_DETAILS.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     VALUE(IS_ALERT) TYPE  CHAR1
*"  TABLES
*"      T_SELECT STRUCTURE  RSSELECT OPTIONAL
*"      T_DATA STRUCTURE  /SKN/S_SW_01_01_VERS_DETAILS OPTIONAL
*"----------------------------------------------------------------------
  DATA_SINGLE: DURATION_UNIT  /SKN/E_SW_DURATION_UNIT,
               BACKDAYS       INT4,
               LANGU          DDLANGUAGE.    " Language
  DATA_MULTY: DURATION      /SKN/E_SW_DURATION,
* VRSD
              OBJTYPE	      VERSOBJTYP,
              OBJNAME	      VERSOBJNAM,
              VERSNO        VERSNO,
              KORRNUM	      VERSKORRNO,
              AUTHOR        VERSUSER,
              DATUM	        VERSDATE,
              ZEIT          VERSTIME,
              LOEKZ	        VERSLOEKZ,
              ORIGIN        VERSORIGIN,
              VERSMODE      VERSMODE,
              LASTVERSNO    VERSLAST,
              FIRSTVERSN    VERSFIRST,
              RELS          VERSRELS,
              KEYLEN        VERSKEYLEN,
              TABLEN        VERSTABLEN,
              DEFVERSNO     DEFVERSNO,
              ARCHIVED      VRS_ARCHVD,
* E070
              TRKORR        TRKORR,         " Request/task
              TRSTATUS      TRSTATUS,       " Status: R-Released, D-Modifiable...
              TRFUNCTION    TRFUNCTION,     " Type: K-Workbench, W-Custom., T-Transport of Copies, R-Repair, S-Develop
              TARSYSTEM     TR_TARGET,
              KORRDEV       TRCATEG,        " Request Category: CUST-Client Specific, SYST-Repository/Cross-Client
              AS4USER       TR_AS4USER,     " Owner
              AS4DATE       AS4DATE,        " Date
              AS4TIME       AS4TIME,        " Time
              STRKORR       STRKORR,        " Higher-Level Request
* E071
              AS4POS        DDPOSITION,     " Line Item
              PGMID         PGMID,          " Program ID
              OBJECT        TROBJTYPE,      " Object Type
              OBJ_NAME      TROBJ_NAME.     " Object Name
*              state_color   /skn/e_sw_state_color,
  DATA: LS_DATA LIKE LINE OF T_DATA,
        LT_DATA LIKE TABLE OF LS_DATA.
  DATA: LS_MAIN_REQ TYPE /SKN/S_SW_TRKOR.
  DATA: LS_VRSD TYPE /SKN/S_SW_01_01_VRSD_DETAILS.
  DATA: LT_VRSD TYPE TABLE OF /SKN/S_SW_01_01_VRSD_DETAILS.
  DATA: LS_QUEUE TYPE TMSBUFFER,
        LT_QUEUE LIKE TABLE OF LS_QUEUE.
  DATA: LV_TRKORR TYPE TRKORR.
  LV_DURATION_UNIT = 'H'.
  LV_BACKDAYS      = 1.
  LV_LANGU         = 'E'.
  SELECT_SINGLE: BACKDAYS,
                 DURATION_UNIT,
                 LANGU.
  SELECT_MULTY: DURATION,
* VRSD
                VERSNO,     " Version Number
                KORRNUM,    " Transport number
                AUTHOR,     " Version Author
                DATUM,      " Date
                ZEIT,       " Time
                LOEKZ,      "
                ORIGIN,     "
                VERSMODE,   "
                LASTVERSNO, " Version Number
                FIRSTVERSN, " Version Number
                RELS,       "
                KEYLEN,
                TABLEN,
                DEFVERSNO,
                ARCHIVED,
* E070
                TRKORR,      " Transport Number
                TRSTATUS,
                TRFUNCTION,
                TARSYSTEM,
                KORRDEV,
                AS4DATE,
                AS4USER,
                AS4TIME,
                STRKORR,     " Higher-Level Request
* E071
                AS4POS,      " Line Item
                OBJECT,      " Obejct Type
                OBJ_NAME,    " Object Name
                PGMID.
*                state_color.
  DATA: IS_GENERAL(1) TYPE C.
  DATA: DATE_FROM LIKE SY-DATUM ,
        BACKDAYS  TYPE I .
  DATA: SY_TABIX LIKE SY-TABIX,
        FROM_TABIX LIKE SY-TABIX.
  DATA: ENDDATE LIKE SY-DATUM,
        ENDTIME LIKE SY-UZEIT.
  DATA: TIME_DIFF TYPE  INT4 .
  DATA: IS_OUT(1) TYPE C.
  DATA: MAX_RC TYPE TRRETCODE,
        MAX_RC_N(4) TYPE N.
  DATA: SY_DATLO LIKE SY-DATLO ,
        SY_TIMLO LIKE SY-TIMLO .
  LV_DURATION_UNIT = 'H'.
  LV_BACKDAYS      = 1.
  LV_LANGU         = 'E'.
  SELECT_SINGLE: BACKDAYS,
                 DURATION_UNIT.
  "--- Run Clood Mode -----
  DATA_SINGLE: SW_DEST RFCDEST.             .
  SELECT_SINGLE: SW_DEST.
  IF LV_SW_DEST IS NOT INITIAL.
    CALL FUNCTION '/SKN/FC_SW_01_01_VERS_DETAILS'
      IMPORTING
        IS_ALERT = IS_ALERT
      TABLES
        T_SELECT = T_SELECT
        T_DATA   = T_DATA.
  ENDIF.
  CHECK LV_SW_DEST IS INITIAL.
  "--- Run Clood Mode -----
  SY_DATLO = SY-DATUM .        " Appl Server's Date
  SY_TIMLO = SY-UZEIT.
  IF R_AS4DATE[] IS NOT INITIAL.
    LOOP AT R_AS4DATE INTO RS_AS4DATE.
      RS_DATUM-SIGN   = RS_AS4DATE-SIGN.
      RS_DATUM-OPTION = RS_AS4DATE-OPTION.
      RS_DATUM-LOW    = RS_AS4DATE-LOW.
      RS_DATUM-HIGH   = RS_AS4DATE-HIGH.
      APPEND RS_DATUM TO R_DATUM.
    ENDLOOP.
  ELSEIF R_DATUM[] IS INITIAL.
    RS_DATUM-SIGN   = 'I'.
    RS_DATUM-OPTION = 'GE'.
    DATE_FROM       = SY_DATLO - LV_BACKDAYS.
    RS_DATUM-LOW    = DATE_FROM.
    APPEND RS_DATUM TO R_DATUM.
  ENDIF.
*--- Retrieve data
  CLEAR IS_ALERT .
  REFRESH: T_DATA, LT_DATA .
  SELECT *
    FROM E070 AS E
    INNER JOIN E071 AS D ON E~TRKORR EQ D~TRKORR
    INNER JOIN E07T AS T ON E~TRKORR EQ T~TRKORR
    INTO CORRESPONDING FIELDS OF TABLE T_DATA
    WHERE E~TRKORR     IN R_TRKORR               " Request/Task
    AND   E~TRSTATUS   IN R_TRSTATUS             " Status:Modifiable, Released
    AND   E~TRFUNCTION IN R_TRFUNCTION           " Type of request/task:Workbench, Customizing
    AND   E~TARSYSTEM  IN R_TARSYSTEM            " Tarnsport Target
    AND   E~KORRDEV    IN R_KORRDEV              " Category
    AND   E~AS4DATE    IN R_DATUM                " Date
    AND   E~AS4TIME    IN R_AS4TIME              " Time
    AND   E~AS4USER    IN R_AS4USER              " Request/task owner
    AND   D~PGMID      IN R_PGMID                " Program ID
    AND   D~OBJECT     IN R_OBJECT               " Object Type
    AND   D~OBJ_NAME   IN R_OBJ_NAME             " Object Name
    AND   T~LANGU      EQ LV_LANGU.              " Language
*  LOOP AT t_data.
*    sy_tabix = sy-tabix .
*
*    max_rc = t_data-maxrc.
*    CASE t_data-maxrc. " MAX_RC.
*      WHEN   '0008' .
*        t_data-state_color = 'R'.
*        t_data-state_icon = state_icon_red .
*      WHEN  '0012'.
*        t_data-state_color = 'R'.
*        t_data-state_icon = '@AG@'. "ICON_ALERT
*      WHEN   '0000'.
*        t_data-state_color ='G'.
*        t_data-state_icon = state_icon_green.
*      WHEN   '0004'.
*        t_data-state_color ='Y'.
*        t_data-state_icon = state_icon_yellow.
*      WHEN ''.
*        t_data-state_icon = '@BZ@'.  "ICON_LED_INACTIVE "'@HM@'. "ICON_PS_MILESTONE
*    ENDCASE.
*    MODIFY t_data INDEX sy_tabix.
*  ENDLOOP.
*
*  DELETE t_data WHERE state_color NOT IN r_state_color.
*-- Fill Duration Value
  LOOP AT T_DATA.
    SY_TABIX = SY-TABIX.
    T_DATA-DURATION_UNIT = LV_DURATION_UNIT.
    CALL FUNCTION '/SKN/F_SW_GET_TIME_DIFF'
      EXPORTING
        D_FROM      = T_DATA-AS4DATE
        T_FROM      = T_DATA-AS4TIME
        D_TO        = SY_DATLO
        T_TO        = SY_TIMLO
        TIME_UNIT   = LV_DURATION_UNIT
      IMPORTING
        TIME_DIFF   = TIME_DIFF
      EXCEPTIONS
        WRONG_VALUE = 1
        OTHERS      = 2.
    IF SY-SUBRC = 0.
      T_DATA-DURATION = TIME_DIFF .
    ELSE.
      T_DATA-DURATION = '999999'.
    ENDIF.
    MODIFY T_DATA INDEX SY_TABIX .
  ENDLOOP.
  DELETE T_DATA WHERE DURATION NOT IN R_DURATION.
*--- Check Alert Information
  READ TABLE T_DATA INDEX 1.
  CHECK NOT SY-TFILL  IS INITIAL .
  IS_ALERT = 'X' .
ENDFUNCTION.
