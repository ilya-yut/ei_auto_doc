  FUNCTION /SKN/F_SW_01_01_SM21.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     VALUE(IS_ALERT) TYPE  CHAR1
*"  TABLES
*"      T_SELECT STRUCTURE  RSSELECT OPTIONAL
*"      T_DATA STRUCTURE  /SKN/S_SW_01_01_SM21 OPTIONAL
*"----------------------------------------------------------------------
DATA_MULTY: DURATION   /SKN/E_SW_DURATION,   "From NOW to Job Start Time point -in duration units
            DATUM    DATUM,
            INSTANCENAME MSNAME2,
            SLGLTRM  RSLGTERM,
            SLGUSER  RSLGUSER,
            SLGTC    TCODE,
            SLGREPNA PROGRAM_ID,
            SLGMAND  MANDT,
            SLGMODE  SAPMODE,
            STATE_COLOR /SKN/E_SW_STATE_COLOR,
            SLGDATE  DATUM,
            SLGTIME  UZEIT,
            WP_TYPE  CHAR4,  "  RSLGWPTYPE,
            DEVCLASS DEVCLASS,
            AREA     RSLGAREA,
            SUBID    RSLGSUBID,
            TEXT     TEXT.
DATA_SINGLE: LANGU LANGU,
             NUMBER SY-TABIX,
             SUBRC SY-SUBRC,
             MANAGE_IN_UTC CHAR1,
             DURATION_UNIT  /SKN/E_SW_DURATION_UNIT,
             BACKDAYS  INT4,
             DEST RFCDEST.
DATA: LS_SYSLOG TYPE RSLGENTR_NEW,
      LT_SYSLOG LIKE TABLE OF LS_SYSLOG.
DATA: LV_POS TYPE I,
      LV_WRAPCOUNT TYPE I,
      LV_FILTER_DATETIME_FROM TYPE  RSLGTIME,
      LV_FILTER_DATETIME_TO TYPE  RSLGTIME,
      LV_INSTANCENAME TYPE  MSNAME2.
DATA: LS_SERVER_LIST TYPE MSXXLIST,
      LT_SERVER_LIST LIKE TABLE OF LS_SERVER_LIST.
DATA: LS_DATA LIKE LINE OF T_DATA,
      LT_DATA LIKE TABLE OF LS_DATA.
DATA : SY_TABIX LIKE SY-TABIX .
DATA :   SY_DATLO LIKE SY-DATLO ,
         SY_TIMLO LIKE SY-TIMLO .
DATA : TIME_DIFF TYPE  INT4 .
DATA: LV_SLGPROC TYPE RSLGPID.
DATA: LV_AREA LIKE  TSL1D-AREA,
      LV_SUBID LIKE  TSL1D-SUBID,
      LV_CACHE_ENTRY TYPE  RSLGTCACHE,
      LV_TERMINAL TYPE CHAR20. "   RSLG_TERMINAL.
DATA : LV_DATE_FROM LIKE SY-DATUM,
       LV_DATE_TO LIKE SY-DATUM.
DATA: LV_NO_SERVER_LIST(1) TYPE C.
CHECK  SY-SAPRL >= '740'. " Check ABAP version (>= 7.40)
***
****-- Fill Selection Option Tables
***
   SELECT_MULTY: DURATION,
                 DATUM,
                 INSTANCENAME,
                 SLGLTRM,
                 SLGUSER,
                 SLGTC,
                 SLGREPNA,
                 SLGMAND,
                 SLGMODE,
                 STATE_COLOR,
                 SLGDATE,
                 SLGTIME,
                 WP_TYPE,
                 DEVCLASS,
                 AREA,
                 SUBID,
                 TEXT.
  LV_DURATION_UNIT = 'M'.
  LV_BACKDAYS = 0.
  SELECT_SINGLE: DURATION_UNIT,
                 BACKDAYS,
                 MANAGE_IN_UTC,
                 DEST.
  DATA_SINGLE: SW_DEST RFCDEST.             .
  SELECT_SINGLE: SW_DEST.
 SET_SY_TIME LV_MANAGE_IN_UTC SY_DATLO SY_TIMLO .
 TIME_SHIFT SY_DATLO SY_TIMLO . " TIME_SHIFT parameter
 READ TABLE R_DATUM INTO RS_DATUM INDEX 1.
 IF SY-SUBRC IS INITIAL.
   LV_DATE_FROM = RS_DATUM-LOW.
   LV_DATE_TO = RS_DATUM-HIGH.
 ELSE.
   LV_DATE_FROM = SY_DATLO - LV_BACKDAYS .
 ENDIF.
*--- Retrieve data
  CLEAR IS_ALERT .
  REFRESH: T_DATA , LT_DATA.
  CALL FUNCTION 'TH_SERVER_LIST'
    DESTINATION LV_SW_DEST
*   EXPORTING
*     SERVICES              = 255
*     SYSSERVICE            = 0
*     ACTIVE_SERVER         = 1
*     SUBSYSTEM_AWARE       = 1
    TABLES
      LIST                  = LT_SERVER_LIST
*     LIST_IPV6             =
    EXCEPTIONS
      NO_SERVER_LIST        = 1
      OTHERS                = 2.
  IF SY-SUBRC <> 0.
    CASE SY-SUBRC.
      WHEN 1.  " NO_SERVER_LIST
        LV_NO_SERVER_LIST = 'X'.
      WHEN OTHERS.
        EXIT.
    ENDCASE.
  ENDIF.
  IF LV_NO_SERVER_LIST IS NOT INITIAL. " Add fictive Item
    LS_SERVER_LIST-NAME = LV_SW_DEST.
    APPEND LS_SERVER_LIST TO LT_SERVER_LIST.
  ENDIF.
  LOOP AT LT_SERVER_LIST INTO LS_SERVER_LIST.
    IF SY-TABIX  > 1.
      EXIT.
    ENDIF.
    REFRESH: LT_SYSLOG.
      LV_POS = 0.
      LV_WRAPCOUNT = 0.
      "concatenate  sy-datum sy-uzeit into lv_FILTER_DATETIME_FROM.
      CONCATENATE  LV_DATE_FROM '000000' INTO LV_FILTER_DATETIME_FROM.
      CLEAR LV_FILTER_DATETIME_TO.
      IF LV_DATE_TO IS NOT INITIAL.
        CONCATENATE  LV_DATE_TO '235959' INTO LV_FILTER_DATETIME_TO.
      ENDIF.
      LV_INSTANCENAME = LS_SERVER_LIST-NAME.
    CALL FUNCTION 'RSLG_READ_SLG'
      DESTINATION LV_SW_DEST
      EXPORTING
        IM_POS                        = LV_POS
        IM_WRAPCOUNT                  = LV_WRAPCOUNT
        IM_FILTER_DATETIME_FROM       = LV_FILTER_DATETIME_FROM
        IM_FILTER_DATETIME_TO         = LV_FILTER_DATETIME_TO
        IM_INSTANCENAME               = LV_INSTANCENAME
*    IMPORTING
*       EX_POS                        =
*       EX_WRAPCOUNT                  =
*       EX_MAXSIZE                    =
      TABLES
        T_SYSLOG_ENTRIES              = LT_SYSLOG.
   LOOP AT LT_SYSLOG INTO LS_SYSLOG.
     MOVE-CORRESPONDING LS_SYSLOG TO LS_DATA.
     LS_DATA-INSTANCENAME = LV_INSTANCENAME.
     LS_DATA-SLGDATE = LS_DATA-SLGDATTIM+0(8).
     LS_DATA-SLGTIME = LS_DATA-SLGDATTIM+8(6).
     LV_SLGPROC = LS_SYSLOG-SLGPROC.
     PERFORM SM21_SYSLOG_TYPE USING LV_SLGPROC
                              CHANGING LS_DATA-PROCESSID LS_DATA-WP_TYPE.
     LV_AREA  = LS_SYSLOG-SLGTYPE+1(2).
     LV_SUBID = LS_SYSLOG-SLGTYPE+3(1).
     CALL FUNCTION 'RSLG_TAB_CACHE_ACCESS_ALV'
       DESTINATION LV_SW_DEST
       EXPORTING
         AREA              = LV_AREA
         SUBID             = LV_SUBID
       IMPORTING
         CACHE_ENTRY       = LV_CACHE_ENTRY.
      LS_DATA-STATE_COLOR = LV_CACHE_ENTRY-MONBEW(1).
      CALL FUNCTION '/SKN/F_SW_GET_STATE_ICON'
        EXPORTING
          STATE_COLOR       = LS_DATA-STATE_COLOR
        IMPORTING
          STATE_ICON        = LS_DATA-STATE_ICON         .
     MOVE-CORRESPONDING LV_CACHE_ENTRY TO LS_DATA.
     LV_TERMINAL = LS_DATA-SLGLTRM.
     PERFORM SM21_SYSLOG_TEXT USING LV_CACHE_ENTRY
                                    LS_DATA-SLGDATA
                                    LS_DATA-SLGPROC
                                    LS_DATA-SLGMODE
                                    LV_TERMINAL
                              CHANGING LS_DATA-ERRNO
                                       LS_DATA-ERRORNAME
                                       LS_DATA-TEXT.
     APPEND LS_DATA TO LT_DATA.
   ENDLOOP.
  ENDLOOP.
  DELETE LT_DATA WHERE STATE_COLOR NOT IN R_STATE_COLOR.
  DELETE LT_DATA WHERE INSTANCENAME NOT IN R_INSTANCENAME.
  DELETE LT_DATA WHERE SLGLTRM NOT IN R_SLGLTRM.
  DELETE LT_DATA WHERE SLGUSER NOT IN R_SLGUSER.
  DELETE LT_DATA WHERE SLGTC NOT IN R_SLGTC.
  DELETE LT_DATA WHERE SLGREPNA NOT IN R_SLGREPNA.
  DELETE LT_DATA WHERE SLGMAND NOT IN R_SLGMAND.
  DELETE LT_DATA WHERE SLGMODE NOT IN R_SLGMODE.
  DELETE LT_DATA WHERE STATE_COLOR NOT IN R_STATE_COLOR.
  DELETE LT_DATA WHERE SLGDATE NOT IN R_SLGDATE.
  DELETE LT_DATA WHERE SLGTIME NOT IN R_SLGTIME.
  DELETE LT_DATA WHERE WP_TYPE NOT IN R_WP_TYPE.
  DELETE LT_DATA WHERE DEVCLASS NOT IN R_DEVCLASS.
  DELETE LT_DATA WHERE AREA NOT IN R_AREA.
  DELETE LT_DATA WHERE SUBID NOT IN R_SUBID.
  DELETE LT_DATA WHERE TEXT NOT IN R_TEXT.
*-- Fill Duration Value
 LOOP AT LT_DATA INTO LS_DATA.
   SY_TABIX = SY-TABIX.
   LS_DATA-DURATION_UNIT = LV_DURATION_UNIT.
    CALL FUNCTION '/SKN/F_SW_GET_TIME_DIFF'
        EXPORTING
          D_FROM            = LS_DATA-SLGDATE
          T_FROM            = LS_DATA-SLGTIME
          D_TO              = SY_DATLO
          T_TO              = SY_TIMLO
          TIME_UNIT         = LV_DURATION_UNIT
        IMPORTING
          TIME_DIFF         = TIME_DIFF
        EXCEPTIONS
          WRONG_VALUE       = 1
          OTHERS            = 2    .
      IF SY-SUBRC = 0.
        LS_DATA-DURATION = TIME_DIFF .
      ELSE.
        LS_DATA-DURATION = '999999'.
      ENDIF.
   MODIFY LT_DATA FROM LS_DATA INDEX SY_TABIX .
 ENDLOOP .
 DELETE LT_DATA WHERE DURATION NOT IN R_DURATION.
T_DATA[] =  LT_DATA[].
*--- Check Alert Information
 READ TABLE T_DATA INDEX 1.
 CHECK NOT SY-TFILL  IS INITIAL .
 IS_ALERT = 'X' .
ENDFUNCTION.
