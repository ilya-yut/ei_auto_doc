FUNCTION /SKN/F_SW_01_01_SM13_UPD_STAT.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     VALUE(IS_ALERT) TYPE  CHAR1
*"  TABLES
*"      T_SELECT STRUCTURE  RSSELECT OPTIONAL
*"      T_DATA STRUCTURE  /SKN/S_SW_01_01_SM13_UPD_STAT OPTIONAL
*"----------------------------------------------------------------------
DATA : LV_ALERT TYPE  CHAR1.
DATA: LV_VB_ACTIVE TYPE VBSTAT1,
      LV_INFOTEXT TYPE  /SKN/E_SW_TEXT_60.
DATA: LS_DATA LIKE LINE OF T_DATA.
DATA_MULTY: VB_ACTIVE VBSTAT1,
            INFOTEXT /SKN/E_SW_TEXT_60.
SELECT_MULTY: VB_ACTIVE,
              INFOTEXT.
DATA_SINGLE:   SW_DEST RFCDEST.
SELECT_SINGLE: SW_DEST.
  CALL FUNCTION 'FUNCTION_EXISTS'
    DESTINATION  LV_SW_DEST
    EXPORTING
      FUNCNAME           = '/SKN/F_SW_01_GET_UPDATE_STATE'
    EXCEPTIONS
      FUNCTION_NOT_EXIST = 1
      COMMUNICATION_FAILURE = 11
      SYSTEM_FAILURE        = 12
      OTHERS             =   9.
  IF SY-SUBRC <> 0.
    EXIT.
  ENDIF.
  CALL FUNCTION '/SKN/F_SW_01_GET_UPDATE_STATE'
    DESTINATION  LV_SW_DEST
    IMPORTING
      VB_ACTIVE       = LV_VB_ACTIVE
      INFOTEXT        = LV_INFOTEXT.
  LS_DATA-VB_ACTIVE = LV_VB_ACTIVE.
  LS_DATA-INFOTEXT = LV_INFOTEXT.
  IF LV_VB_ACTIVE IN R_VB_ACTIVE AND
     LV_INFOTEXT  IN R_INFOTEXT.
    APPEND LS_DATA TO T_DATA.
    IS_ALERT = 'X'.
  ENDIF.
ENDFUNCTION.
