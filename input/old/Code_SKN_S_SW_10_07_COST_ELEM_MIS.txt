            KTOPL      = LV_KTOPL           " Chart of Accounts

            <FS_VAL> = LV_TXT50.
            MODIFY T_DATA.
            UNASSIGN <FS_VAL>.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.
    MODIFY T_DATA[] FROM  T_DATA.
  ENDLOOP.
