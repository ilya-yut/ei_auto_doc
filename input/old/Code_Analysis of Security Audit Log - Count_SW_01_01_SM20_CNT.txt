  FUNCTION /SKN/F_SW_01_01_SM20_CNT.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     VALUE(IS_ALERT) TYPE  CHAR1
*"  TABLES
*"      T_SELECT STRUCTURE  RSSELECT OPTIONAL
*"      T_DATA STRUCTURE  /SKN/S_SW_01_01_SM20_CNT OPTIONAL
*"----------------------------------------------------------------------
DATA : LV_ALERT TYPE  CHAR1.
DATA : LS_DATA TYPE /SKN/S_SW_01_01_SM20,
       LT_DATA LIKE TABLE OF LS_DATA.
DATA: LS_T_DATA LIKE LINE OF T_DATA,
      LT_T_DATA LIKE TABLE OF LS_T_DATA.
DATA : LV_CNT TYPE I.
DATA_MULTY: REQ_CNT /SKN/E_SW_CNT.
SELECT_MULTY: REQ_CNT.
   REFRESH T_DATA.
   CALL FUNCTION '/SKN/F_SW_01_01_SM20'
    IMPORTING
       IS_ALERT       = LV_ALERT
     TABLES
       T_SELECT       = T_SELECT
       T_DATA         = LT_DATA.
    IS_ALERT = LV_ALERT.
    LOOP AT LT_DATA INTO LS_DATA.
      MOVE-CORRESPONDING LS_DATA TO LS_T_DATA.
      LS_T_DATA-REQ_CNT = 1.
      COLLECT LS_T_DATA INTO LT_T_DATA.
    ENDLOOP.
    LOOP AT LT_T_DATA INTO LS_T_DATA.
      IF LS_T_DATA-REQ_CNT IN R_REQ_CNT.
        APPEND LS_T_DATA TO T_DATA.
      ENDIF.
    ENDLOOP.
*--- Check Alert Information
  READ TABLE T_DATA INDEX 1.
  CHECK NOT SY-TFILL  IS INITIAL .
  IS_ALERT = 'X' .
ENDFUNCTION.
