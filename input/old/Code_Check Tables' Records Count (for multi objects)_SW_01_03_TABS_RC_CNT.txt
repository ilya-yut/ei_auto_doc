FUNCTION /SKN/F_SW_01_03_TABS_REC_COUNT .
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     VALUE(IS_ALERT) TYPE  CHAR1
*"  TABLES
*"      T_SELECT STRUCTURE  RSSELECT OPTIONAL
*"      T_DATA STRUCTURE  /SKN/S_SW_01_03_TAB_REC_COUNT OPTIONAL
*"----------------------------------------------------------------------
*data: R_TAB type range of TABNAME.
*data: RS_TAB like line of R_TAB.
"--- Add Parameter for Compare State & Attr Compare String
DATA: LV_DEST TYPE RFCDEST,
      LV_TAB  TYPE TABNAME,
      LV_LANGU TYPE LANGU.
"------------------------------------------
DATA : ROWCOUNT TYPE INT4.
DATA : SY_TABIX LIKE SY-TABIX .
DATA: QUERY_TABLE TYPE TABNAME,
      LS_OPTIONS  TYPE  RFC_DB_OPT,
      LT_OPTIONS LIKE TABLE OF LS_OPTIONS. " ,
DATA: BEGIN OF LS_TAB,
        TABNAME TYPE  TABNAME,
      END OF LS_TAB.
DATA: LT_TAB LIKE TABLE OF LS_TAB. " ,
DATA: LS_DATA LIKE LINE OF T_DATA.
**-- Fill Selection Option Tables
DATA_MULTY: TAB  TABNAME.
SELECT_SINGLE: DEST,
               LANGU.
IF LV_LANGU IS INITIAL.
  LV_LANGU = SY-LANGU.
ENDIF.
SELECT_MULTY: TAB.
 CLEAR IS_ALERT .
 REFRESH T_DATA.
 "---- Prepare Input Parameters (tables)
 "--- Fill Condition Criteria
 CHECK R_TAB[] IS NOT INITIAL.
 REFRESH LT_TAB.
 CLEAR LS_TAB.
* loop at R_TAB into RS_TAB.
*   ls_TAB-TABNAME = RS_TAB-low.
*   append ls_TAB to lt_TAB.
* endloop.
  "--- Run Cloud Mode -----
  DATA_SINGLE: SW_DEST RFCDEST.             .
  SELECT_SINGLE: SW_DEST.
  IF LV_SW_DEST IS NOT INITIAL.
    CALL FUNCTION '/SKN/FC_SW_01_03_TABS_REC_CNT'
      IMPORTING
        IS_ALERT       = IS_ALERT
      TABLES
        T_SELECT       = T_SELECT
        T_DATA         = T_DATA.
  ENDIF.
  CHECK LV_SW_DEST IS INITIAL.
"--- Run Cloud Mode -----
 SELECT *
   FROM DD02L
   INTO CORRESPONDING FIELDS OF TABLE LT_TAB
   WHERE TABNAME IN R_TAB
     AND TABCLASS NOT IN ('INTTAB' , 'APPEND').
   CHECK LT_TAB[] IS NOT INITIAL.
*  "--- Retrieve Table data Count
   "--- Where Condition Prepare
   REFRESH T_DATA.
   LOOP AT LT_TAB INTO LS_TAB.
     CLEAR LS_DATA.
     LV_TAB = LS_TAB-TABNAME.
     QUERY_TABLE = LV_TAB.
     REFRESH LT_OPTIONS.
     CLEAR ROWCOUNT.
     CALL FUNCTION '/SKN/F_SW_RFC_TABLE_REC_CNT'
        DESTINATION        LV_DEST
       EXPORTING
         QUERY_TABLE       = QUERY_TABLE
       IMPORTING
         ROWCOUNT          = ROWCOUNT
       TABLES
         OPTIONS           = LT_OPTIONS
      EXCEPTIONS
        TABLE_NOT_AVAILABLE       = 1
        TABLE_WITHOUT_DATA        = 2
        OPTION_NOT_VALID          = 3
        OTHERS                    = 4.
     IF SY-SUBRC = 0.
       "--- Fill Output Table
       LS_DATA-TAB = LV_TAB.
       LS_DATA-RECORDS = ROWCOUNT.
       LS_DATA-DEST = LV_DEST.
       CALL FUNCTION '/SKN/F_SW_TABLE_DESC'
         EXPORTING
           TABNAME        = LS_DATA-TAB
           LANGU          = LV_LANGU
         IMPORTING
           TAB_DESC       = LS_DATA-TAB_DESC
         EXCEPTIONS
           WRONG_TABLE                       = 1
           NO_DESCRIPTION_FOR_LANGUAGE       = 2
           OTHERS                            = 3.
       IF SY-SUBRC <> 0.
       ENDIF.
       APPEND LS_DATA TO T_DATA.
     ENDIF.
  ENDLOOP.
 DESCRIBE TABLE T_DATA LINES SY-TFILL .
 CHECK NOT SY-TFILL  IS INITIAL .
 IS_ALERT = 'X' .
ENDFUNCTION.
